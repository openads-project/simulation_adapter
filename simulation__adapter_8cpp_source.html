50template <typename T>
51void SimulationAdapter::declareAndLoadParameter(const std::string& name,
52 T& param,
53 const std::string& description,
54 const bool add_to_auto_reconfigurable_params,
55 const bool is_required,
56 const bool read_only,
57 const std::optional<double>& from_value,
58 const std::optional<double>& to_value,
59 const std::optional<double>& step_value,
60 const std::string& additional_constraints) {
61 rcl_interfaces::msg::ParameterDescriptor param_desc;
62 param_desc.description = description;
63 param_desc.additional_constraints = additional_constraints;
64 param_desc.read_only = read_only;
65
66 auto type = rclcpp::ParameterValue(param).get_type();
67
68 if (from_value.has_value() && to_value.has_value()) {
69 if constexpr (std::is_integral_v<T>) {
70 rcl_interfaces::msg::IntegerRange range;
71 T step = static_cast<T>(step_value.has_value() ? step_value.value() : 1);
72 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value())).set__step(step);
73 param_desc.integer_range = {range};
74 } else if constexpr (std::is_floating_point_v<T>) {
75 rcl_interfaces::msg::FloatingPointRange range;
76 T step = static_cast<T>(step_value.has_value() ? step_value.value() : 1.0);
77 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value())).set__step(step);
78 param_desc.floating_point_range = {range};
79 } else {
80 RCLCPP_WARN(this->get_logger(), "Parameter type of parameter '%s' does not support specifying a range", name.c_str());
81 }
82 }
83
84 this->declare_parameter(name, type, param_desc);
85
86 try {
87 param = this->get_parameter(name).get_value<T>();
88 std::stringstream ss;
89 ss << "Loaded parameter '" << name << "': ";
90 if constexpr (is_vector_v<T>) {
91 ss << "[";
92 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "]");
93 } else {
94 ss << param;
95 }
96 RCLCPP_INFO_STREAM(this->get_logger(), ss.str());
97 } catch (rclcpp::exceptions::ParameterUninitializedException&) {
98 if (is_required) {
99 RCLCPP_FATAL_STREAM(this->get_logger(), "Missing required parameter '" << name << "', exiting");
100 exit(EXIT_FAILURE);
101 } else {
102 std::stringstream ss;
103 ss << "Missing parameter '" << name << "', using default value: ";
104 if constexpr (is_vector_v<T>) {
105 ss << "[";
106 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "]");
107 } else {
108 ss << param;
109 }
110 RCLCPP_WARN_STREAM(this->get_logger(), ss.str());
111 this->set_parameters({rclcpp::Parameter(name, rclcpp::ParameterValue(param))});
112 }
113 }
114
115 if (add_to_auto_reconfigurable_params) {
116 std::function<void(const rclcpp::Parameter&)> setter = [&param](const rclcpp::Parameter& p) { param = p.get_value<T>(); };
117 auto_reconfigurable_params_.push_back(std::make_tuple(name, setter));
118 }
119}

127rcl_interfaces::msg::SetParametersResult SimulationAdapter::parametersCallback(const std::vector<rclcpp::Parameter>& parameters) {
128 for (const auto& param : parameters) {
129 for (auto& auto_reconfigurable_param : auto_reconfigurable_params_) {
130 if (param.get_name() == std::get<0>(auto_reconfigurable_param)) {
131 std::get<1>(auto_reconfigurable_param)(param);
132 RCLCPP_INFO(this->get_logger(), "Reconfigured parameter '%s' to: %s", param.get_name().c_str(),
133 param.value_to_string().c_str());
134 break;
135 }
136 }
137 }
138
139 rcl_interfaces::msg::SetParametersResult result;
140 result.successful = true;
141
142 return result;
143}

149void SimulationAdapter::setup() {
150 // initialize tf2 buffer, listener and static broadcaster
151 tf2_buffer_ = std::make_unique<tf2_ros::Buffer>(this->get_clock());
152 tf2_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf2_buffer_);
153 static_br_tf_ = std::make_unique<tf2_ros::StaticTransformBroadcaster>(this);
154
155 map_info_ = "";
156
157 // parameters client to map server for setting map server's parameters
158 map_server_parameters_client_ = std::make_shared<rclcpp::AsyncParametersClient>(this, map_server_name_);
159 using namespace std::chrono_literals;
160 while (!map_server_parameters_client_->wait_for_service(1s)) {
161 if (!rclcpp::ok()) {
162 RCLCPP_FATAL(this->get_logger(), "Interrupted while waiting for the map server ('%s') parameter service, shutting down",
163 map_server_name_.c_str());
164 rclcpp::shutdown();
165 return;
166 }
167 RCLCPP_WARN(this->get_logger(), "Waiting for map server ('%s') parameter service ...", map_server_name_.c_str());
168 }
169 RCLCPP_INFO(this->get_logger(), "Connected to map server ('%s') parameter service", map_server_name_.c_str());
170
171 // create a callback for dynamic parameter configuration
172 parameters_callback_ =
173 this->add_on_set_parameters_callback(std::bind(&SimulationAdapter::parametersCallback, this, std::placeholders::_1));
174
175 // setup subscriber for input topics
176 // Use transient_local (latching) QoS so the node receives the map info even
177 // when it starts after the publisher has already sent the single startup message.
178 rclcpp::QoS qosLatching = rclcpp::QoS(rclcpp::KeepLast(1));
179 qosLatching.transient_local();
180 qosLatching.reliable();
181 reentrant_callback_group_ = this->create_callback_group(rclcpp::CallbackGroupType::Reentrant);
182
183 rclcpp::SubscriptionOptions reentrant_subscription_options;
184 reentrant_subscription_options.callback_group = reentrant_callback_group_;
185
186 sub_map_info_ = this->create_subscription<sm::String>(
187 kInputMapInfoTopic, qosLatching, std::bind(&SimulationAdapter::mapInfoCallback, this, std::placeholders::_1),
188 reentrant_subscription_options);
189 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", sub_map_info_->get_topic_name());
190
191 sub_ego_data_ = this->create_subscription<pm::EgoData>(
192 kInputEgoDataTopic, 1, std::bind(&SimulationAdapter::egoDataCallback, this, std::placeholders::_1),
193 reentrant_subscription_options);
194 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", sub_ego_data_->get_topic_name());
195
196 sub_object_list_ = this->create_subscription<pm::ObjectList>(
197 kInputObjectListTopic, 1, std::bind(&SimulationAdapter::objectListCallback, this, std::placeholders::_1),
198 reentrant_subscription_options);
199 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", sub_object_list_->get_topic_name());
200
201 sub_trajectory_ = this->create_subscription<tp::Trajectory>(
202 kInputTrajectoryTopic, 1, std::bind(&SimulationAdapter::trajectoryCallback, this, std::placeholders::_1),
203 reentrant_subscription_options);
204 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", sub_trajectory_->get_topic_name());
205
206 if (publish_vehicle_frame_tf_) {
207 tf_init_timer_ = this->create_wall_timer(500ms, std::bind(&SimulationAdapter::initializeVehicleFrameTransform, this),
208 reentrant_callback_group_);
209 RCLCPP_INFO(this->get_logger(), "Started timer to initialize transformation from '%s' to '%s'", fixed_frame_id_.c_str(),
210 vehicle_frame_id_.c_str());
211 } else {
212 RCLCPP_INFO(this->get_logger(), "Static TF publication from '%s' to '%s' is disabled", simulation_vehicle_frame_id_.c_str(),
213 vehicle_frame_id_.c_str());
214 }
215
216 // set up publisher for output topics
217 pub_ego_data_ = this->create_publisher<pm::EgoData>(kEgoDataTopic, 1);
218 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", pub_ego_data_->get_topic_name());
219
220 if (publish_ego_odometry_) {
221 pub_ego_odometry_ = this->create_publisher<nav_msgs::msg::Odometry>(kEgoOdometryTopic, 1);
222 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", pub_ego_odometry_->get_topic_name());
223 }
224
225 if (publish_ego_vehicle_state_) {
226 pub_ego_vehicle_state_ = this->create_publisher<pm::ObjectState>(kEgoVehicleStateTopic, 1);
227 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", pub_ego_vehicle_state_->get_topic_name());
228 }
229
230 if (publish_ego_imu_) {
231 pub_ego_imu_ = this->create_publisher<sensor_msgs::msg::Imu>(kEgoImuTopic, 1);
232 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", pub_ego_imu_->get_topic_name());
233 }
234
235 pub_object_list_ = this->create_publisher<pm::ObjectList>(kObjectListTopic, 1);
236 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", pub_object_list_->get_topic_name());
237
238 pub_object_list_fixed_ = this->create_publisher<pm::ObjectList>(kObjectListFixedTopic, 1);
239 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", pub_object_list_fixed_->get_topic_name());
240
241 // logging info
242 RCLCPP_INFO(this->get_logger(), "simulation_adapter is running...");
243}

250void SimulationAdapter::mapInfoCallback(const sm::String::ConstSharedPtr& msg) {
251 if (msg->data == map_info_) return;
252 map_info_ = msg->data;
253
254 std::string lanelet_map_name;
255 if (set_ll2_map_) {
256 // convert simulation name to lanelet map name
257 std::string simulation_map_name = map_info_;
258
259 // find simulation_map_name in simulation_maps_
260 auto it = std::find(simulation_maps_.begin(), simulation_maps_.end(), simulation_map_name);
261 if (it == simulation_maps_.end()) {
262 RCLCPP_ERROR(this->get_logger(), "Simulation map name '%s' not found in the list of supported maps",
263 simulation_map_name.c_str());
264 return;
265 }
266
267 // Get lanelet map for simulation map name
268 size_t index = std::distance(simulation_maps_.begin(), it);
269 if (index >= lanelet_files_.size()) {
270 RCLCPP_ERROR(this->get_logger(),
271 "No lanelet file at index %zu for map '%s'. Check that 'maps.simulation_maps' and "
272 "'maps.lanelet_files' have the same length in params.yml.",
273 index, simulation_map_name.c_str());
274 return;
275 }
276 lanelet_map_name = lanelet_files_[index];
277
278 // change map by setting map server parameters
279 map_server_parameters_client_->set_parameters(
280 {rclcpp::Parameter("map_filepath", lanelet_map_name), rclcpp::Parameter("map_frame_id", fixed_frame_id_)},
281 [this](std::shared_future<std::vector<rcl_interfaces::msg::SetParametersResult>> future) {
282 auto results = future.get();
283 for (const auto& result : results) {
284 if (!result.successful) RCLCPP_ERROR(this->get_logger(), "Failed to set parameter: %s", result.reason.c_str());
285 }
286 RCLCPP_INFO(this->get_logger(), "Finished setting map server parameters");
287 });
288 }

291void SimulationAdapter::egoDataCallback(const pm::EgoData::ConstSharedPtr& msg) {
292 auto timeout = rclcpp::Duration::from_seconds(0.1);
293
294 // Transform ego_data pose into fixed_frame_id. The reference point is adjusted below under the assumption that
295 // simulation_vehicle_frame_id and vehicle_frame_id differ only by a longitudinal offset and share orientation.
296 pm::EgoData ego_data;
297
298 gm::TransformStamped to_map_tf;
299 try {
300 to_map_tf = tf2_buffer_->lookupTransform(fixed_frame_id_, msg->header.frame_id, msg->header.stamp, timeout);
301 } catch (tf2::TransformException& ex) {
302 RCLCPP_WARN(this->get_logger(), "Transformation from '%s' to '%s' is not available.", msg->header.frame_id.c_str(),
303 fixed_frame_id_.c_str());
304 return;
305 }
306 tf2::doTransform(*msg, ego_data, to_map_tf);
307
308 // Move the reported position from simulation_vehicle_frame_id to vehicle_frame_id. This is only a pose/reference-point
309 // correction; velocity and acceleration are kept as provided by the simulation.
310 tf2::Quaternion vehicle_orientation;
311 tf2::fromMsg(perception_msgs::object_access::getOrientation(ego_data.state), vehicle_orientation);
312 const tf2::Vector3 vehicle_offset(simulation_vehicle_frame_id_to_vehicle_frame_id_, 0.0, 0.0);
313 const tf2::Vector3 offset_in_map = tf2::quatRotate(vehicle_orientation, vehicle_offset);
314
315 perception_msgs::object_access::setX(ego_data, perception_msgs::object_access::getX(ego_data.state) + offset_in_map.x());
316 perception_msgs::object_access::setY(ego_data, perception_msgs::object_access::getY(ego_data.state) + offset_in_map.y());
317 perception_msgs::object_access::setZ(ego_data, perception_msgs::object_access::getZ(ego_data.state) + offset_in_map.z());
318
319 ego_data.state.reference_point.value = pm::ObjectReferencePoint::REAR_AXLE_GROUND;
320 ego_data.state.reference_point.translation_to_geometric_center.x = -simulation_vehicle_frame_id_to_vehicle_frame_id_;
321 ego_data.state.reference_point.translation_to_geometric_center.z = msg->height / 2.0;
322 perception_msgs::object_access::setZ(ego_data, perception_msgs::object_access::getZ(ego_data.state) - msg->height / 2.0);
323
324 // add planned trajectory to ego_data if exists
325 tp::Trajectory trajectory_planned_snapshot;
326 {
327 std::lock_guard<std::mutex> lock(trajectory_planned_mutex_);
328 trajectory_planned_snapshot = trajectory_planned_;
329 }
330
331 int n = trajectory_planning_msgs::trajectory_access::getSamplePointSize(trajectory_planned_snapshot);
332 if (n > 0) {
333 // clear current trajectory
334 ego_data.trajectory_planned.clear();
335
336 // initialize state
337 pm::ObjectState object_state;
338 perception_msgs::object_access::initializeState(object_state, 1);
339 object_state.reference_point = ego_data.state.reference_point;
340
341 // update trajectory state
342 perception_msgs::object_access::setStandstill(
343 object_state, trajectory_planning_msgs::trajectory_access::getStandstill(trajectory_planned_snapshot));
344
345 for (int i = 0; i < n; i++) {
346 // update header stamp
347 object_state.header = trajectory_planned_snapshot.header;
348 const double time = trajectory_planning_msgs::trajectory_access::getT(trajectory_planned_snapshot, i);
349 object_state.header.stamp = rclcpp::Time(trajectory_planned_snapshot.header.stamp) + rclcpp::Duration::from_seconds(time);
350
351 perception_msgs::object_access::setX(object_state,
352 trajectory_planning_msgs::trajectory_access::getX(trajectory_planned_snapshot, i));
353 perception_msgs::object_access::setY(object_state,
354 trajectory_planning_msgs::trajectory_access::getY(trajectory_planned_snapshot, i));
355 perception_msgs::object_access::setVelLon(
356 object_state, trajectory_planning_msgs::trajectory_access::getV(trajectory_planned_snapshot, i));
357 perception_msgs::object_access::setAccLon(
358 object_state, trajectory_planning_msgs::trajectory_access::getA(trajectory_planned_snapshot, i));
359 perception_msgs::object_access::setYaw(
360 object_state, trajectory_planning_msgs::trajectory_access::getTheta(trajectory_planned_snapshot, i));
361 ego_data.trajectory_planned.push_back(object_state);
362 }
363 }
364
365 // publish ego data in fixed_frame_id
366 pub_ego_data_->publish(ego_data);
367
368 if (publish_ego_odometry_) {
369 // Odometry is published in map/<vehicle_frame_id> convention: pose in map, twist in the body frame.
370 nav_msgs::msg::Odometry ego_odometry;
371 ego_odometry.header.frame_id = fixed_frame_id_;
372 ego_odometry.header.stamp = ego_data.state.header.stamp;
373 ego_odometry.child_frame_id = vehicle_frame_id_;
374 ego_odometry.pose.pose.position.x = perception_msgs::object_access::getX(ego_data.state);
375 ego_odometry.pose.pose.position.y = perception_msgs::object_access::getY(ego_data.state);
376 ego_odometry.pose.pose.position.z = perception_msgs::object_access::getZ(ego_data.state);
377 ego_odometry.pose.pose.orientation = perception_msgs::object_access::getOrientation(ego_data.state);
378 ego_odometry.pose.covariance = perception_msgs::object_access::getPoseWithCovariance(ego_data.state).covariance;
379 ego_odometry.twist.twist.linear = perception_msgs::object_access::getVelocity(ego_data.state);
380 ego_odometry.twist.twist.angular.z = perception_msgs::object_access::getYawRate(ego_data.state);
381 pub_ego_odometry_->publish(ego_odometry);
382 }
383
384 if (publish_ego_vehicle_state_) {
385 pm::ObjectState ego_vehicle_state;
386 perception_msgs::object_access::initializeState(ego_vehicle_state, pm::EGO::MODEL_ID);
387 ego_vehicle_state.header = ego_data.state.header;
388 if (perception_msgs::object_access::hasSteeringAngleAck(ego_data.state.model_id)) {
389 perception_msgs::object_access::setSteeringAngleAck(ego_vehicle_state,
390 perception_msgs::object_access::getSteeringAngleAck(ego_data.state));
391 }
392 if (perception_msgs::object_access::hasSteeringAngleRateAck(ego_data.state.model_id)) {
393 perception_msgs::object_access::setSteeringAngleRateAck(
394 ego_vehicle_state, perception_msgs::object_access::getSteeringAngleRateAck(ego_data.state));
395 }
396 pub_ego_vehicle_state_->publish(ego_vehicle_state);
397 }
398
399 if (publish_ego_imu_) {
400 sensor_msgs::msg::Imu ego_imu;
401 ego_imu.header.stamp = ego_data.state.header.stamp;
402 ego_imu.header.frame_id = vehicle_frame_id_;
403 ego_imu.orientation_covariance[0] = -1.0;
404 ego_imu.angular_velocity_covariance[0] = -1.0;
405 ego_imu.linear_acceleration = perception_msgs::object_access::getAcceleration(ego_data.state);
406 pub_ego_imu_->publish(ego_imu);
407 }
408}

410void SimulationAdapter::objectListCallback(const pm::ObjectList::ConstSharedPtr& msg) {
411 auto timeout = rclcpp::Duration::from_seconds(0.1);
412
413 // Option A: transform object_list to fixed_frame_id
414 pm::ObjectList msg_object_list_fixed;
415
416 gm::TransformStamped to_map_tf;
417 try {
418 to_map_tf = tf2_buffer_->lookupTransform(fixed_frame_id_, msg->header.frame_id, msg->header.stamp, timeout);
419 } catch (tf2::TransformException& ex) {
420 RCLCPP_WARN(this->get_logger(), "Transformation from '%s' to '%s' is not available.", msg->header.frame_id.c_str(),
421 fixed_frame_id_.c_str());
422 return;
423 }
424 tf2::doTransform(*msg, msg_object_list_fixed, to_map_tf);
425
426 // publish object list in fixed_frame_id
427 pub_object_list_fixed_->publish(msg_object_list_fixed);
428
429 // Option B: transform object list to vehicle_frame_id_
430 pm::ObjectList msg_object_list;
431
432 gm::TransformStamped to_vehicle_frame_tf;
433 try {
434 to_vehicle_frame_tf = tf2_buffer_->lookupTransform(vehicle_frame_id_, msg->header.frame_id, msg->header.stamp, timeout);
435 } catch (tf2::TransformException& ex) {
436 RCLCPP_WARN(this->get_logger(), "Skipping object list transform from '%s' to '%s': %s", msg->header.frame_id.c_str(),
437 vehicle_frame_id_.c_str(), ex.what());
438 return;
439 }
440 tf2::doTransform(*msg, msg_object_list, to_vehicle_frame_tf);
441
442 // publish object_list in vehicle_frame_id
443 pub_object_list_->publish(msg_object_list);
444}

446void SimulationAdapter::initializeVehicleFrameTransform() {
447 /* set up a transformation link between fixed_frame_id and vehicle_frame_id
448
449 / utm_<zone> \
450 / \
451 / static \ static (published by lanelet2_map_server)
452 / (published by \
453 / rspecific simulation \
454 simulation_fixed_frame_id fixed_frame_id
455 |
456 dynamic (published by specific simulation)
457 |
458 v
459 simulation_vehicle_frame_id ---static---> vehicle_frame_id
460 */
461
462 auto timezero = tf2::TimePointZero;
463
464 try {
465 // check if desired transformation is already defined
466 tf2_buffer_->lookupTransform(vehicle_frame_id_, fixed_frame_id_, timezero);
467 if (tf_init_timer_ && !tf_init_timer_->is_canceled()) {
468 tf_init_timer_->cancel();
469 RCLCPP_INFO(this->get_logger(), "Static transformation from '%s' to '%s' is now available", vehicle_frame_id_.c_str(),
470 fixed_frame_id_.c_str());
471 }
472 return;
473 } catch (const tf2::TransformException&) {
474 RCLCPP_DEBUG(this->get_logger(), "Transformation from '%s' to '%s' not yet available, retrying ...", fixed_frame_id_.c_str(),
475 vehicle_frame_id_.c_str());
476
477 // step 1: simulation_fixed_frame_id -> fixed_frame_id
478 try {
479 tf2_buffer_->lookupTransform(fixed_frame_id_, simulation_fixed_frame_id_, timezero);
480 } catch (const tf2::TransformException& ex) {
481 RCLCPP_WARN(this->get_logger(),
482 "Transformation from '%s' to '%s' is not available. Should be provided using a shared parent "
483 "utm frame.",
484 simulation_fixed_frame_id_.c_str(), fixed_frame_id_.c_str());
485 RCLCPP_WARN(this->get_logger(), "\tSkipped ...");
486 return;
487 }
488
489 // step 2: simulation_fixed_frame_id -> simulation_vehicle_frame_id
490 try {
491 tf2_buffer_->lookupTransform(simulation_vehicle_frame_id_, simulation_fixed_frame_id_, timezero);
492 } catch (const tf2::TransformException& ex) {
493 RCLCPP_WARN(this->get_logger(), "\tTransformation from '%s' to '%s' not available", simulation_fixed_frame_id_.c_str(),
494 simulation_vehicle_frame_id_.c_str());
495 RCLCPP_WARN(this->get_logger(), "\tSkipped ...");
496 return;
497 }
498
499 // step 3: simulation_vehicle_frame_id -> vehicle_frame_id
500 try {
501 tf2_buffer_->lookupTransform(vehicle_frame_id_, simulation_vehicle_frame_id_, timezero);
502 } catch (const tf2::TransformException& ex) {
503 RCLCPP_WARN(this->get_logger(), "\tTransformation from '%s' to '%s' is not available: %s",
504 simulation_vehicle_frame_id_.c_str(), vehicle_frame_id_.c_str(), ex.what());
505
506 // publish static transformation from simulation_vehicle_frame_id to vehicle_frame_id
507 gm::TransformStamped ego_vehicle_to_vehicle_frame;
508 ego_vehicle_to_vehicle_frame.header.stamp = this->get_clock()->now();
509 ego_vehicle_to_vehicle_frame.header.frame_id = simulation_vehicle_frame_id_;
510 ego_vehicle_to_vehicle_frame.child_frame_id = vehicle_frame_id_;
511
512 ego_vehicle_to_vehicle_frame.transform.translation.x = simulation_vehicle_frame_id_to_vehicle_frame_id_;
513 ego_vehicle_to_vehicle_frame.transform.translation.y = 0.0;
514 ego_vehicle_to_vehicle_frame.transform.translation.z = 0.0;
515
516 tf2::Quaternion q;
517 q.setRPY(0, 0, 0);
518 ego_vehicle_to_vehicle_frame.transform.rotation.x = q.x();
519 ego_vehicle_to_vehicle_frame.transform.rotation.y = q.y();
520 ego_vehicle_to_vehicle_frame.transform.rotation.z = q.z();
521 ego_vehicle_to_vehicle_frame.transform.rotation.w = q.w();
522
523 static_br_tf_->sendTransform(ego_vehicle_to_vehicle_frame);
524 RCLCPP_INFO(this->get_logger(), "\tTransformation from '%s' to '%s' was published", simulation_vehicle_frame_id_.c_str(),
525 vehicle_frame_id_.c_str());
526 }
527 }

530void SimulationAdapter::trajectoryCallback(const tp::Trajectory::ConstSharedPtr& msg) {
531 if (msg->type_id != trajectory_planning_msgs::msg::DRIVABLE::TYPE_ID) {
532 RCLCPP_WARN(this->get_logger(),
533 "Invalid trajectory type, planned trajectory states are only filled for trajectories of type DRIVABLE");
534 return;
535 }
536
537 // transform trajectory to fixed_frame_id frame
538 try {
539 auto transformed_trajectory = tf2_buffer_->transform(*msg, fixed_frame_id_, tf2::durationFromSec(0.01));
540 std::lock_guard<std::mutex> lock(trajectory_planned_mutex_);
541 trajectory_planned_ = std::move(transformed_trajectory);
542 } catch (tf2::TransformException& ex) {
543 RCLCPP_WARN(this->get_logger(), "Trajectory could not be transformed from '%s' to '%s'", msg->header.frame_id.c_str(),
544 fixed_frame_id_.c_str());
545 return;
546 }
547}
